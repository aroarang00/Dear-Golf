function CompanionAddModal({ visible, onClose, onAdd, currentCompanions }) {
  const [name, setName] = useState('');
  const existingNames = (currentCompanions || []).map(c => c.name);
  const suggestions = FRIENDS_DATA.filter(f => !existingNames.includes(f.nickname));

  const handleAdd = (n) => {
    const trimmed = (n || name).trim();
    if (!trimmed) return;
    if (existingNames.includes(trimmed)) {
      Alert.alert('?뚮┝', '?대? 異붽????숇컲?먯삁??);
      return;
    }
    onAdd({ name: trimmed, isMe: false });
    setName('');
    onClose();
  };

  return (
    <Modal visible={visible} transparent animationType="slide" onRequestClose={onClose}>
      <KeyboardAvoidingView behavior={Platform.OS === 'ios' ? 'padding' : undefined} style={{ flex: 1 }}>
        <View style={mS.mask}>
          <TouchableOpacity style={{ flex: 1 }} activeOpacity={1} onPress={onClose} />
          <View style={[mS.sheet, { maxHeight: '50%' }]}>
            <View style={{ alignItems: 'center', paddingVertical: 10 }}>
              <View style={mS.handle} />
            </View>
            <View style={{ padding: 20, paddingTop: 0 }}>
              <Text style={mS.title}>?숇컲??異붽?</Text>
              <View style={{ flexDirection: 'row', gap: 8, marginTop: 8 }}>
                <TextInput
                  style={[mS.input, { flex: 1 }]}
                  placeholder="?대쫫 ?먮뒗 ?됰꽕???낅젰"
                  placeholderTextColor={C.warmGrayLight}
                  value={name}
                  onChangeText={setName}
                  autoFocus
                />
                <TouchableOpacity
                  style={{ backgroundColor: C.charcoal, borderRadius: 10,
                    paddingHorizontal: 16, justifyContent: 'center' }}
                  onPress={() => handleAdd()}>
                  <Text style={{ fontFamily: F.sys, fontSize: 13, color: C.butter }}>異붽?</Text>
                </TouchableOpacity>
              </View>
              {suggestions.length > 0 && (
                <View style={{ marginTop: 14 }}>
                  <Text style={{ fontFamily: F.sys, fontSize: 10, color: C.warmGrayLight,
                    letterSpacing: 1.5, marginBottom: 8 }}>移쒓뎄 紐⑸줉?먯꽌 ?좏깮</Text>
                  <ScrollView horizontal showsHorizontalScrollIndicator={false}>
                    <View style={{ flexDirection: 'row', gap: 8 }}>
                      {suggestions.map(f => (
                        <TouchableOpacity key={f.id}
                          onPress={() => handleAdd(f.nickname)}
                          style={{ backgroundColor: C.paleSky + '44', borderRadius: 20,
                            paddingHorizontal: 14, paddingVertical: 8,
                            borderWidth: 0.5, borderColor: C.paleSky }}>
                          <Text style={{ fontFamily: F.sys, fontSize: 12, color: C.charcoal }}>
                            {f.nickname}
                          </Text>
                        </TouchableOpacity>
                      ))}
                    </View>
                  </ScrollView>
                </View>
              )}
            </View>
          </View>
        </View>
      </KeyboardAvoidingView>
    </Modal>
  );
}
